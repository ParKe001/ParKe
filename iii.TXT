repeat task.wait() until game:IsLoaded()
if workspace.DistributedGameTime < 3 then
    task.wait(3 - workspace.DistributedGameTime)
end
Players = game:GetService("Players")
plr = Players.LocalPlayer
rs = game:GetService("RunService")
run = true
con = {}
chr = nil
hum = nil
hea = nil
rot = nil
table.insert(con, rs.Heartbeat:Connect(function()
    cc = plr.Character
    if not cc or not cc.Parent then
        chr, hum, hea, rot = nil, nil, nil, nil
        return
    end
    chr = cc
    if not hum or hum.Parent ~= chr then
        hum = chr:FindFirstChildOfClass("Humanoid")
    end
    if not hea or hea.Parent ~= chr then
        hea = chr:FindFirstChild("Head")
    end
    if not rot or rot.Parent ~= chr then
        rot = chr.PrimaryPart
            or chr:FindFirstChild("HumanoidRootPart")
            or chr:FindFirstChild("Torso")
    end
end))

arg = true

table.insert(con, rs.Heartbeat:Connect(function(dt)
    if hum and arg and
        ((hum:GetState() == Enum.HumanoidStateType.Ragdoll) or
         (hum:GetState() == Enum.HumanoidStateType.FallingDown)) then
        hum:ChangeState(Enum.HumanoidStateType.GettingUp)
    end
end))
