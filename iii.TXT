local Cln = cloneref or clone_ref or clonereference or clone_reference
local NCl = function(Obj)
    if Cln then
        local Suc, Res = pcall(Cln, Obj)
        return Suc and Res or Obj
    end
    return Obj
end

local function GetR(Char)
    return Char and (Char.PrimaryPart or Char:FindFirstChild("HumanoidRootPart"))
end

local function RemD(Obj)
    if Obj then
        pcall(function()
            Obj:Remove()
        end)
    end
end

local ESPD = {}

local function GetE(Plr)
    local D = ESPD[Plr]
    if not D then
        D = Drawing.new("Text")
        D.Size = 16
        D.Center = true
        D.Outline = true
        D.Visible = false
        ESPD[Plr] = D
    end
    return D
end

local function ClrE(Plr)
    local D = ESPD[Plr]
    if D then
        RemD(D)
        ESPD[Plr] = nil
    end
end

table.insert(Cnn, Plr.PlayerRemoving:Connect(ClrE))

local function UseS(SkiN, KeyC)
    local Ski = LCha and SkiL:FindFirstChild(SkiN)
    if not Ski then
        return
    end

    local FraS = Ski:FindFirstChild("Frame_Status")
    if not FraS then
        return
    end

    local FraC = FraS:FindFirstChild("Frame_CD")
    if not FraC or FraC.Size.Y.Scale ~= 0 then
        return
    end

    local Com = LCha:FindFirstChild("Communicate")
    if Com and Com:IsA("RemoteEvent") then
        Com:FireServer({
            ["Key"] = KeyC, ["State"] = Enum.UserInputState.Begin
        })
    end
end

task.spawn(function()
    while true do

        end

            UseS("Skill2", Enum.KeyCode.Two)
        end

        if Cfg.AutoSecondSkill then
            UseS("Skill3", Enum.KeyCode.Three)
        end

        task.wait(0.075)
    end
end)

table.insert(Cnn, RunS.Heartbeat:Connect(function(DelT)
    if Cfg.FOVChanger then
        Cam.FieldOfView = Cfg.FOVValue
    end

    if Cfg.Speed and LHum and LRoo then
        local MovD = LHum.MoveDirection
        if MovD.Magnitude > 0 then
            LRoo.CFrame += MovD * Cfg.SpeedAmount
        end
    end

    if LHum and Cfg.AntiRagdoll and ((LHum:GetState() == Enum.HumanoidStateType.Ragdoll) or (LHum:GetState() == Enum.HumanoidStateType.FallingDown)) then
        LHum:ChangeState(Enum.HumanoidStateType.GettingUp)
    end

    if Cfg.Attaching then
        if not LRoo or not Loc:GetAttribute("InDuel") then
            return
        end

        local CloR = nil
        local BigD = 185
        for _,Pl in ipairs(Plr:GetPlayers()) do
            if Pl == Loc then
                continue
            end

            if not Pl:GetAttribute("InDuel") then
                continue
            end

            local Cha = Pl.Character
            if not Cha then
                continue
            end

            local Roo = GetR(Cha)
            if not Roo then
                continue
            end

            local Mag = (LRoo.Position - Roo.Position).Magnitude
            if Mag < BigD then
                CloR = Roo
                BigD = Mag
            end
        end

        if CloR then
            local CloC = CloR.CFrame
            LRoo.CFrame = CFrame.lookAt((CloC * CFrame.new(0, Cfg.UpOffset, Cfg.BackOffset)).Position, CloC.Position)
            if Cfg.BodyVelocity then
                local BodV = LRoo:FindFirstChildOfClass("BodyVelocity")
                if BodV then
                    BodV.MaxForce = Vector3.new(5000, 5000, 5000)
                end
            end
        end
    elseif Cfg.BodyVelocity then
        local BodV = LRoo and LRoo:FindFirstChildOfClass("BodyVelocity")
        if BodV then
            BodV.MaxForce = Vector3.zero
        end
    end

    for _,Pl in ipairs(Plr:GetPlayers()) do
        if Pl == Loc then
            continue
        end

        local ESP = GetE(Pl)
        local IsV = nil

        local Cha = Pl.Character
        local Hea = Cha and Cha:FindFirstChild("Head")
        local Hum = Cha and Cha:FindFirstChildOfClass("Humanoid")

        if Cfg.ESPEnabled and Hea and Hum and Hum.Health > 0 then
            local Scr, OnS = Cam:WorldToViewportPoint(Hea.Position)
            if OnS then
                IsV = true

                local TexP = {}
                if Cfg.ESPShowNames then
                    table.insert(TexP, Pl.DisplayName)
                end

                if Cfg.ESPShowHealth then
                    table.insert(TexP, string.format("[%d 血量]", math.floor(Hum.Health)))
                end

                if LRoo and Cfg.ESPShowDistance then
                    table.insert(TexP, string.format("[%d 米]", math.floor((LRoo.Position - Hea.Position).Magnitude)))
                end

                if #TexP > 0 then
                    ESP.Text = table.concat(TexP, "\n")
                    ESP.Position = Vector2.new(Scr.X - 40, Scr.Y - 40)
                    ESP.Color = Cfg.ESPColor
                    ESP.Visible = true
                else
                    ESP.Visible = false
                end
            end
        end

        if not IsV then
            ESP.Visible = false
        end
    end

    for Pl in next, ESPD do
        if Pl.Parent ~= Plr then
            ClrE(Pl)
        end
    end
end))
